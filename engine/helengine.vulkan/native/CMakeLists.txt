cmake_minimum_required(VERSION 3.25)
project(helengine_vulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan 1.1 REQUIRED)
find_program(HELENGINE_GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(HELENGINE_VULKAN_SHADER_OUTPUT_ROOT "${CMAKE_BINARY_DIR}/shaders" CACHE PATH "Compiled portable Vulkan shader directory.")
option(HELENGINE_VULKAN_BUILD_TESTS "Build portable Vulkan CPU and ownership tests." OFF)
option(HELENGINE_VULKAN_STATIC_MESH_BATCHING "Batch compatible Static entity meshes." OFF)
set(HELENGINE_VULKAN_STATIC_MESH_BATCHING_BUDGET_MIB "256" CACHE STRING "Additional resident static geometry budget in MiB.")
option(HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING "Batch compatible small current-frame meshes." OFF)
set(HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_MAX_VERTICES "1024" CACHE STRING "Maximum referenced vertices per original dynamic input.")
set(HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_BUDGET_MIB "4" CACHE STRING "Reusable transient dynamic geometry budget in MiB.")
foreach(limit HELENGINE_VULKAN_STATIC_MESH_BATCHING_BUDGET_MIB HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_MAX_VERTICES HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_BUDGET_MIB)
    string(LENGTH "${${limit}}" limit_length)
    if(NOT "${${limit}}" MATCHES "^[1-9][0-9]*$" OR limit_length GREATER 10 OR "${${limit}}" GREATER 2147483647)
        message(FATAL_ERROR "${limit} must be a positive decimal Int32 integer.")
    endif()
endforeach()
foreach(flag HELENGINE_VULKAN_STATIC_MESH_BATCHING HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING)
    string(TOUPPER "${${flag}}" flag_value)
    if(NOT flag_value MATCHES "^(ON|OFF|TRUE|FALSE|YES|NO|1|0)$")
        message(FATAL_ERROR "${flag} must be a CMake boolean.")
    endif()
endforeach()
add_library(helengine_vulkan STATIC src/vulkan_context.cpp src/vulkan_renderer_2d.cpp src/vulkan_mesh_renderer.cpp src/vulkan_indirect_culler.cpp src/vulkan_indirect_page.cpp)
target_include_directories(helengine_vulkan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(helengine_vulkan PUBLIC cxx_std_20)
target_link_libraries(helengine_vulkan PUBLIC Vulkan::Vulkan)
target_compile_definitions(helengine_vulkan PUBLIC HELENGINE_VULKAN_SHADER_ROOT="${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}")
target_compile_definitions(helengine_vulkan PUBLIC
    HELENGINE_VULKAN_STATIC_MESH_BATCHING=$<BOOL:${HELENGINE_VULKAN_STATIC_MESH_BATCHING}>
    HELENGINE_VULKAN_STATIC_MESH_BATCHING_BUDGET_MIB=${HELENGINE_VULKAN_STATIC_MESH_BATCHING_BUDGET_MIB}
    HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING=$<BOOL:${HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING}>
    HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_MAX_VERTICES=${HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_MAX_VERTICES}
    HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_BUDGET_MIB=${HELENGINE_VULKAN_DYNAMIC_MESH_BATCHING_BUDGET_MIB})
if(MSVC)
    target_compile_options(helengine_vulkan PRIVATE /W4 /we4715)
else()
    target_compile_options(helengine_vulkan PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
endif()
if(WIN32)
    target_compile_definitions(helengine_vulkan PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
foreach(shader frustum_cull.comp quad.vert quad.frag shadow_depth.vert shadow_depth.frag point_shadow_depth.vert point_shadow_depth.frag)
    if(shader MATCHES "^(shadow_depth|point_shadow_depth)\.(vert|frag)$")
        set(input "${CMAKE_MATCH_1}.hlsl")
        set(stage "${CMAKE_MATCH_2}")
        if(stage STREQUAL "vert")
            set(entry VS)
        else()
            set(entry PS)
        endif()
        set(arguments -D -V --target-env vulkan1.2 --hlsl-iomap -S "${stage}" -e "${entry}")
    else()
        set(input "${shader}")
        set(arguments -V --target-env vulkan1.1)
    endif()
    add_custom_command(OUTPUT "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/${shader}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}"
        COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" ${arguments} "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${input}" -o "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/${shader}.spv"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${input}" VERBATIM)
    list(APPEND compiled_shaders "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/${shader}.spv")
endforeach()
add_custom_target(helengine_vulkan_shaders DEPENDS ${compiled_shaders})
add_dependencies(helengine_vulkan helengine_vulkan_shaders)
if(HELENGINE_CORE_CPP_ROOT AND EXISTS "${HELENGINE_CORE_CPP_ROOT}/RenderCommandList2D.hpp")
    add_library(helengine_vulkan_bridge STATIC src/vulkan_render_manager_2d.cpp src/vulkan_render_manager_3d.cpp src/vulkan_render_manager_3d_snapshots.cpp src/vulkan_render_manager_3d_properties.cpp src/vulkan_render_manager_3d_batching.cpp src/vulkan_render_manager_3d_material.cpp src/vulkan_render_manager_3d_lights.cpp src/vulkan_render_manager_3d_shadows.cpp)
    target_include_directories(helengine_vulkan_bridge PUBLIC "${HELENGINE_CORE_CPP_ROOT}")
    target_link_libraries(helengine_vulkan_bridge PUBLIC helengine_vulkan)
endif()
if(HELENGINE_VULKAN_BUILD_TESTS)
    enable_testing()
    foreach(stage vert frag)
        add_custom_command(OUTPUT "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/indirect_page.${stage}.spv"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}"
            COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "${CMAKE_CURRENT_SOURCE_DIR}/tests/indirect_page.${stage}" -o "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/indirect_page.${stage}.spv"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/indirect_page.${stage}" VERBATIM)
        list(APPEND indirect_test_shaders "${HELENGINE_VULKAN_SHADER_OUTPUT_ROOT}/indirect_page.${stage}.spv")
    endforeach()
    add_custom_target(helengine_vulkan_indirect_test_shaders DEPENDS ${indirect_test_shaders})
    add_executable(helengine_vulkan_indirect_page_test tests/indirect_page_test.cpp)
    target_link_libraries(helengine_vulkan_indirect_page_test PRIVATE helengine_vulkan)
    add_dependencies(helengine_vulkan_indirect_page_test helengine_vulkan_indirect_test_shaders)
    add_test(NAME vulkan_indirect_page COMMAND helengine_vulkan_indirect_page_test)
    set_tests_properties(vulkan_indirect_page PROPERTIES LABELS gpu)
    add_library(helengine_vulkan_frame_trace_fixture STATIC tests/frame_trace_other_translation_unit.cpp)
    target_include_directories(helengine_vulkan_frame_trace_fixture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    add_executable(helengine_vulkan_frame_trace_test tests/frame_trace_test.cpp)
    target_include_directories(helengine_vulkan_frame_trace_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(helengine_vulkan_frame_trace_test PRIVATE helengine_vulkan_frame_trace_fixture)
    add_test(NAME vulkan_frame_trace_enabled COMMAND helengine_vulkan_frame_trace_test enabled)
    add_test(NAME vulkan_frame_trace_disabled COMMAND helengine_vulkan_frame_trace_test disabled)

    add_executable(helengine_vulkan_prepared_bindings_test tests/prepared_bindings_test.cpp)
    target_link_libraries(helengine_vulkan_prepared_bindings_test PRIVATE helengine_vulkan)
    add_test(NAME vulkan_prepared_bindings COMMAND helengine_vulkan_prepared_bindings_test)

    add_executable(helengine_vulkan_context_contract_test tests/context_contract_test.cpp)
    target_link_libraries(helengine_vulkan_context_contract_test PRIVATE helengine_vulkan)
    add_test(NAME vulkan_context_contract COMMAND helengine_vulkan_context_contract_test)
    set_tests_properties(vulkan_context_contract PROPERTIES LABELS gpu)
    foreach(test indirect_bounds indirect_range_plan mesh_batcher mesh_batch_eligibility uniform_arena_layout material_texture_cache queue_selection material_snapshot extension_policy sparse_snapshot root_material_cache inherited_material_cache prepared_material)
        add_executable(helengine_vulkan_${test}_test "tests/${test}_test.cpp")
        target_link_libraries(helengine_vulkan_${test}_test PRIVATE helengine_vulkan)
        add_test(NAME vulkan_${test} COMMAND helengine_vulkan_${test}_test)
    endforeach()
endif()
