cmake_minimum_required(VERSION 3.16)
project(helengine_gi_native LANGUAGES CXX)
# Directory-local flag changes never change the embedding engine's configuration.
# /RTC checks are incompatible with /O2; preserve /MDd and debug symbols for ABI/debugging.
if(MSVC)
    string(REGEX REPLACE "[-/](Od|Ob0|RTC[1su]+)" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
endif()
add_library(helengine_gi STATIC src/VoxelGi.cpp)
set_target_properties(helengine_gi PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(helengine_gi PUBLIC include)
target_compile_features(helengine_gi PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(helengine_gi PRIVATE /W4 /fp:precise $<$<CONFIG:Debug>:/O2> $<$<CONFIG:Debug>:/Zi>)
else()
    target_compile_options(helengine_gi PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()
option(HELENGINE_GI_TESTS "Build the standalone portable GI tests" ON)
if(HELENGINE_GI_TESTS)
    enable_testing()
    add_executable(helengine_gi_tests tests/VoxelGiTests.cpp)
    target_link_libraries(helengine_gi_tests PRIVATE helengine_gi)
    add_test(NAME helengine_gi_tests COMMAND helengine_gi_tests)
endif()

option(HELENGINE_GI_BENCHMARKS "Build a repeatable standalone GI staging benchmark" OFF)
if(HELENGINE_GI_BENCHMARKS)
    add_executable(helengine_gi_benchmark tests/VoxelGiBenchmark.cpp)
    target_link_libraries(helengine_gi_benchmark PRIVATE helengine_gi)
endif()
